### 4. LLM辅助批量处理工具链

#### Step 1: 提取问题句子
```bash
python3 << 'EOF'
import re
import json

patterns = {
    "他知道": r'他知道[，,]',
    "这不是": r'这不是',
    "不是因为": r'不是因为',
    "沉默了片刻": r'沉默了片刻',
    "深吸一口气": r'深吸一口气'
}

with open('novel-lingse-complete-100-chapters.md', 'r') as f:
    content = f.read()
    
results = []
for pattern_name, pattern in patterns.items():
    matches = list(re.finditer(pattern, content))
    for match in matches:
        # 提取上下文（前后50字）
        start = max(0, match.start() - 50)
        end = min(len(content), match.end() + 50)
        context = content[start:end]
        results.append({
            "pattern": pattern_name,
            "position": match.start(),
            "context": context
        })

with open('ai_taste_sentences.json', 'w') as f:
    json.dump(results, f, ensure_ascii=False, indent=2)

print(f"提取完成: {len(results)}处AI味道")
EOF
```

## 三、技术工具

### 工具1: 提取AI味道句子

```bash
python3 << 'EOF'
import re
import json

# 读取小说
with open('novel-lingse-complete-100-chapters.md', 'r') as f:
    content = f.read()

# 定义AI味道模式
patterns = {
    "他知道": r'他知道[，,：:]',
    "这不是": r'这不是',
    "不是因为": r'不是因为',
    "沉默了片刻": r'沉默了[片一]刻|沉默了一会',
    "深吸一口气": r'深吸[一了]口[凉浊]?气'
}

results = []
for pattern_name, pattern_regex in patterns.items():
    matches = list(re.finditer(pattern_regex, content))
    for match in matches:
        # 提取上下文（前后80字）
        start = max(0, match.start() - 80)
        end = min(len(content), match.end() + 80)
        context = content[start:end]
        
        # 找到所在章节
        chapter_match = re.search(r'## 第(\d+)章', content[:match.start()][::-1])
        chapter = chapter_match.group(1) if chapter_match else "未知"
        
        results.append({
            "pattern": pattern_name,
            "chapter": chapter,
            "position": match.start(),
            "matched_text": match.group(),
            "context": context.replace('\n', ' ')
        })

# 保存结果
with open('ai_taste_sentences.json', 'w', encoding='utf-8') as f:
    json.dump(results, f, ensure_ascii=False, indent=2)

# 统计
stats = {}
for r in results:
    stats[r['pattern']] = stats.get(r['pattern'], 0) + 1

print("=" * 50)
print("AI味道检测完成")
print("=" * 50)
for pattern, count in sorted(stats.items(), key=lambda x: -x[1]):
    print(f"  {pattern}: {count}次")
print(f"\n总计: {len(results)}处")
print(f"已保存到: ai_taste_sentences.json")
EOF
```

### 工具2: 生成修复建议

```bash
# 注意: 需要配置DeepSeek API密钥
python3 << 'EOF'
import json
import os
from openai import OpenAI

# 读取提取的句子
with open('ai_taste_sentences.json', 'r', encoding='utf-8') as f:
    sentences = json.load(f)

# 配置DeepSeek API
client = OpenAI(
    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url="https://api.deepseek.com"
)

# 批量生成建议（分批处理，每批10条）
suggestions = []
batch_size = 10

for i in range(0, len(sentences), batch_size):
    batch = sentences[i:i+batch_size]
    
    prompt = "以下是小说中存在AI味道的句子，请为每一句提供改写建议。\n\n"
    prompt += "要求:\n"
    prompt += "1. 消除AI痕迹，使表达更自然\n"
    prompt += "2. 保持原意和上下文连贯\n"
    prompt += "3. 可以删除、改写或具象化\n\n"
    
    for j, sent in enumerate(batch):
        prompt += f"{j+1}. 【{sent['pattern']}】{sent['context']}\n\n"
    
    prompt += "请以JSON数组格式返回，每项包含: original, suggestion, reason"
    
    response = client.chat.completions.create(
        model="deepseek-chat",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.3
    )
    
    # 解析结果
    result = json.loads(response.choices[0].message.content)
    suggestions.extend(result)
    
    print(f"已处理: {min(i+batch_size, len(sentences))}/{len(sentences)}")

# 保存建议
with open('rewrite_suggestions.json', 'w', encoding='utf-8') as f:
    json.dump(suggestions, f, ensure_ascii=False, indent=2)

print(f"\n✅ 生成完成: {len(suggestions)}条建议")
print("已保存到: rewrite_suggestions.json")
EOF
```
